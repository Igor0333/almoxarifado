-- SELECIONAR NOTA FISCAL DE ENTRADA
SELECT * FROM entrada2
WHERE NOTA LIKE '%053991%'
AND DATA>='2022-03-01'
ORDER BY CONTROLE,ITEM

SELECT * FROM movto
WHERE NOTA='053991'
ORDER BY ITEM

-- BUSCAR PRODUTO MOVIMENTADO POR NOTA FISCAL DE SAIDA
SELECT
B.EMISSAO,A.NOTA,A.PRODUTO,A.DESCR,A.QTDE
FROM
nota2 AS A,nota1 AS B
WHERE
A.NOTA=B.NOTA AND
B.EMISSAO>='2021-01-01' AND
--A.NOTA IN ('004368','004382','004407')
A.PRODUTO='PN0001034'
ORDER BY A.PRODUTO



------------------------------------------------------
-- BUSCAR MOVIMENTAÇÃO DE PRODUTO ESPECÍFICO BEM COMO SALDOS
DECLARE
@DTINI date='2021-08-01',
@DTFIM date='2021-12-31'
SELECT
	B.PRODUTO AS PRODUTO,
	D.DESCRICAO AS DESCRICAO,
	REPLACE(CONVERT(VARCHAR,CAST(SUM(B.QTDE) AS NUMERIC(18,5)), 1),'.',',') AS QTDE_NF_S,
	CASE WHEN E.QTDE_MOVTO_S IS NULL THEN REPLACE(CONVERT(VARCHAR,CAST(0 AS NUMERIC(18,5)), 1),'.',',')
	ELSE REPLACE(CONVERT(VARCHAR,CAST(E.QTDE_MOVTO_S AS NUMERIC(18,5)), 1),'.',',') END AS QTDE_MOV_S,
	CASE WHEN C.QTDE_MOVTO_E IS NULL THEN REPLACE(CONVERT(VARCHAR,CAST(0 AS NUMERIC(18,5)), 1),'.',',')
	ELSE REPLACE(CONVERT(VARCHAR,CAST(C.QTDE_MOVTO_E AS NUMERIC(18,5)), 1),'.',',') END AS QTDE_MOV_E,
	CASE WHEN F.SALDO_INI IS NULL THEN REPLACE(CONVERT(VARCHAR,CAST(0 AS NUMERIC(18,5)), 1),'.',',')
	ELSE REPLACE(CONVERT(VARCHAR,CAST(F.SALDO_INI AS NUMERIC(18,5)), 1),'.',',') END AS SALDO_INICIAL,
	REPLACE(CONVERT(VARCHAR,CAST(
	((CASE WHEN F.SALDO_INI IS NULL THEN 0 ELSE F.SALDO_INI END) + (CASE WHEN C.QTDE_MOVTO_E IS NULL THEN 0 ELSE C.QTDE_MOVTO_E END) +
	CASE WHEN E.QTDE_MOVTO_S IS NULL THEN 0 ELSE  E.QTDE_MOVTO_S END) AS NUMERIC(18,5)), 1),'.',',') AS SALDO_FINAL
	FROM
	FAT00001.dbo.nota1 AS A,
	FAT00001.dbo.nota2 AS B
	LEFT JOIN (SELECT C.PROD,SUM(C.QUANT) AS QTDE_MOVTO_E FROM FAT00001.dbo.movto AS C 
	WHERE C.TIPO='E' AND
	C.DATA >= @DTINI
	AND C.DATA <= @DTFIM
	GROUP BY C.PROD) AS C ON B.PRODUTO=C.PROD
	LEFT JOIN (SELECT C.PROD,SUM(C.QUANT) AS QTDE_MOVTO_S FROM FAT00001.dbo.movto AS C 
	WHERE C.TIPO='S' AND
	C.DATA >= @DTINI
	AND C.DATA <= @DTFIM
	GROUP BY C.PROD) AS E ON B.PRODUTO=E.PROD
	LEFT JOIN (SELECT A.PROD,SUM(A.QUANT) AS SALDO_INI FROM FAT00001.dbo.movto AS A
	WHERE A.DATA < @DTINI GROUP BY A.PROD) AS F ON F.PROD=B.PRODUTO,
	FAT00001.dbo.produto AS D
	WHERE
	A.NOTA=B.NOTA
	AND B.PRODUTO=D.CODIGO
	AND A.STATUSNFE = 'Autorizada'
	AND B.PRODUTO='PN0001034'
	AND A.EMISSAO >= @DTINI
	AND B.CFOP NOT IN ('5.902','6.902','5.923','6.923','1.556','2.556','3.556','1.551','2.551','3.101','3.127','3.551','5.551','6.551','7.551','5.915','6.915','1.916','2.916','1.949','2.949')
	GROUP BY B.PRODUTO,D.DESCRICAO,C.QTDE_MOVTO_E,E.QTDE_MOVTO_S,F.SALDO_INI
	ORDER BY B.PRODUTO
