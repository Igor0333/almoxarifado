-- >> PARA SELECIONAR MOVIMENTACAO ESTOQUE NO MES ANTERIOR, O QUE FAZ DIFERENÇA É O PENULTIMO NÚMERO NA SELEÇÃO DE DATA
SELECT
TIPO,DATA,CONTROLE,NOTA,PROD,DESCRICAO,ITEM,QUANT,OBSSAI
FROM FAT00001.dbo.movto AS A
WHERE
DATA >= DATEADD(s,-1,DATEADD(mm,DATEDIFF(m,0,GETDATE()) -1,0))
AND DATA <= DATEADD(s,-1,DATEADD(mm,DATEDIFF(m,0,GETDATE()),0))
ORDER BY DATA,NOTA,CONTROLE,ITEM
