
DECLARE
@PROD varchar(50)='ADM03208',
@DTINI date='2022-02-01',
@DTFIM date='2022-02-28'

SELECT
A.EMISSAO,A.NOTA,A.RAZAOCLI,A.RAZAOFOR,B.PRODUTO,B.DESCR,B.QTDE,B.CFOP,A.STATUSNFE
FROM
nota1 AS A, nota2 AS B
WHERE
A.NOTA=B.NOTA
AND A.EMISSAO>=@DTINI AND A.EMISSAO<=@DTFIM
AND B.PRODUTO=@PROD

SELECT * 
FROM movto AS A WHERE A.PROD=@PROD
AND A.DATA>=@DTINI AND A.DATA<=@DTFIM
